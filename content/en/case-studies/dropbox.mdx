<Choice
  id="rabin-chunking-advantage"
  objective="content-addressed-chunks"
  prompt="Why does Content-Defined Chunking (Rabin rolling hash) outperform fixed 4 MB chunk offsets when a user inserts a paragraph at the top of a 100 MB file?"
  options={[
    { id: "boundary-resync", text: "Chunk boundaries depend on local byte content rather than absolute file offset, so boundaries re-synchronise right after the insertion and only the first chunk changes", correct: true, feedback: "Right.     { id: "faster-than-memcpy", text: "Rabin hashing uses zero CPU instructions", feedback: "Rolling hashes use slightly more CPU than fixed offsets, trading a bit of client CPU for massive bandwidth and storage savings." },
    { id: "smaller-hashes", text: "Rabin hashes replace SHA-256 as the block key", feedback: "Rabin rolling hashes only decide where to cut chunk boundaries; each chunk is still identified by its cryptographic SHA-256 hash." }
  ]}
