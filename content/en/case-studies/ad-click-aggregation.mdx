    { id: "blind-increment", text: "Execute `UPDATE aggregates SET clicks = clicks + 850 WHERE ad_id = 42 AND window = '12:05'`", feedback: "If the worker crashes after the DB update but before committing its stream offset, the replay will add 850 a second time (1,700)." },
    { id: "insert-no-key", text: "Insert a new row with a random UUID on every flush", feedback: "Replays will insert duplicate rows that inflate `SUM(clicks)`." }
