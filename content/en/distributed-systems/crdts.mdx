<Choice
  id="naive-merge"
  objective="gcounter"
  prompt="Two replicas each hold a single number for a like counter. On sync, each sets its value to the sum of both. What goes wrong?"
  options={[
    { id: "double", text: "Repeated syncs keep adding the same likes again, so the count grows without new likes", correct: true, feedback: "Right. Addition is not idempotent. Syncing twice double-counts, which is why the G-Counter keeps per-replica entries and takes a maximum." },
    { id: "lost", text: "Concurrent likes are lost", feedback: "Summing does not lose likes; it overcounts them on every repeated sync." },
    { id: "nothing", text: "Nothing, addition is commutative", feedback: "Commutativity is not enough. Merge must also be idempotent, and addition is not." },
    { id: "max", text: "It is the same as taking the maximum", feedback: "Taking the maximum of two totals loses the other replica's concurrent likes; summing them double-counts on repeated syncs. The fix is per-replica entries." }
  ]}
/>

</Step>
