<Choice
  id="multiwindow"
  objective="alert-on-burn-rate"
  prompt="Why does the 1-hour burn-rate alert also require the last 5 minutes to exceed the threshold?"
  options={[
    { id: "reset", text: "So the alert resets quickly once the problem is fixed, rather than firing until the bad hour ages out", correct: true, feedback: "Right. The long window gives significance; the short window confirms the burn is still happening now." },
    { id: "faster", text: "So it fires within 5 minutes of any error at all", feedback: "The long window still has to reach 14.4× burn, so a short blip does not page. The short window gates reset, not sensitivity." },
    { id: "cheaper", text: "Because 5-minute queries are cheaper than 1-hour queries", feedback: "Query cost is not the reason. Both windows run; the pair trades detection time against reset time." },
    { id: "sla", text: "Because SLAs are measured in 5-minute intervals", feedback: "The windows come from budget arithmetic, not from any SLA convention." }
  ]}
