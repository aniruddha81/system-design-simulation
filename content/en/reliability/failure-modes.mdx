<Choice
  id="metastable-recovery"
  objective="explain-metastability"
  multiple
  prompt="The database pause that triggered a retry storm ended ten minutes ago, but the service is still overloaded and every request times out. Which actions help it recover? Select all that apply."
  options={[
    { id: "shed", text: "Shed load at the edge, for example admit only a fraction of traffic, then raise it gradually", correct: true, feedback: "Right. Admitted load below capacity lets the queue drain, so requests start succeeding and retries stop." },
    { id: "disable-retries", text: "Push a client or proxy config that disables or budgets retries", correct: true, feedback: "Right. This removes the sustaining effect directly: load falls back to the organic rate." },
    { id: "drop-queue", text: "Drop the queued backlog, whose callers have already timed out", correct: true, feedback: "Right. Work for callers who gave up is pure waste. Serving it first keeps every new request late too." },
    { id: "wait", text: "Wait; the trigger is over, so the system will settle", feedback: "That is exactly what a metastable state does not do. The retry loop sustains the overload without the trigger." },
    { id: "restart-cold", text: "Restart every server at once to start fresh", feedback: "Restarts drop capacity to zero, then bring servers back with cold caches into the same retry load. The loop resumes, often worse." }
  ]}
/>
