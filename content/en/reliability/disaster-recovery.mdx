<Ordering
  id="pitr-runbook"
  objective="protect-against-logical-loss"
  prompt="Put the steps of recovering from a destructive bad deploy in order."
  items={[
    { id: "stop", text: "Stop the damage: roll back the deploy or block the writes that corrupt data" },
    { id: "pick", text: "Find the last good point in time from logs and the deploy timeline" },
    { id: "restore", text: "Restore the most recent snapshot before that point into a separate instance" },
    { id: "replay", text: "Replay archived logs up to just before the bad writes" },
    { id: "verify", text: "Verify the restored data with integrity checks and spot comparisons" },
    { id: "cutover", text: "Reconcile later good writes and shift traffic to the restored database" }
  ]}
  explanation="Stopping the damage first keeps the target still. Restoring into a separate instance keeps the current state available for reconciliation; verification before cutover means you never swap one bad database for another."
/>
