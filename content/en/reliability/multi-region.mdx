<Choice
  id="cart-conflicts"
  objective="resolve-write-conflicts"
  prompt="A shopping cart is active-active in two regions. A user adds an item on their phone (routed to region A) and another on their laptop (routed to region B) in the same second. Which approach keeps both items?"
  options={[
    { id: "crdt", text: "Model the cart as a set CRDT whose replicas merge by union", correct: true, feedback: "Right. Adds commute, so merging the two replicas keeps both items. Removes need care, such as an observed-remove set." },
    { id: "lww", text: "Last-writer-wins on the whole cart object", feedback: "The later write replaces the earlier cart, so one of the two items disappears." },
    { id: "clock", text: "Synchronise both regions' clocks precisely, then use last-writer-wins", feedback: "Better clocks pick the true later write, but last-writer-wins still discards the other update." },
    { id: "ignore", text: "Nothing is needed; replication delivers both writes to both regions", feedback: "Both writes arrive, but each replaces the whole cart. Without a merge rule, one overwrite wins." }
  ]}
